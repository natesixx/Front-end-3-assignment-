        // Initialize rates from API
        async function fetchRates() {
            try {
                const res = await fetch('https://open.er-api.com/v6/latest/USD');
                if(!res.ok) throw new Error('Network error');
                const data = await res.json();
                if(data && data.rates) {
                    currentRates = data.rates;
                    document.getElementById('lastUpdatedText').innerText = 'Updated: Live';
                }
            } catch (e) {
                console.warn('Using offline/fallback rates:', e);
                document.getElementById('lastUpdatedText').innerText = 'Rates: Offline Mode';
            }
            calculateConversion();
            updateQuickGrid();
            updateChart();
        }

        // Calculate and Render Conversion Result
        function calculateConversion() {
            const amount = parseFloat(amountInput.value) || 0;
            const fromRate = currentRates[fromCurr.code] || FALLBACK_RATES[fromCurr.code] || 1;
            const toRate = currentRates[toCurr.code] || FALLBACK_RATES[toCurr.code] || 1;

            // Convert via base (USD)
            const baseAmount = amount / fromRate;
            const finalResult = baseAmount * toRate;

            const unitRate = (1 / fromRate) * toRate;

            // Formatting
            convertedAmountEl.innerText = finalResult.toLocaleString(undefined, {
                minimumFractionDigits: 2,
                maximumFractionDigits: 4
            });
            convertedCodeEl.innerText = toCurr.code;
            conversionFormulaEl.innerText = `1 ${fromCurr.code} = ${unitRate.toFixed(4)} ${toCurr.code}`;
            fromSymbolPrefix.innerText = fromCurr.symbol;
        }

            renderList(toList, toSearch.value, (curr) => {
                toCurr = curr;
                updateDropdownUI('to');
                closeDropdowns();
                calculateConversion();
                updateChart();
            });
        }

                    </div>
                `;
                item.addEventListener('click', () => selectCallback(curr));
                container.appendChild(item);
            });
        }

            toMenu.classList.add('hidden');

                if(!target) return;

        // Charting Logic (Simulated historical data based on current exchange rate)
        function updateChart() {
            const ctx = document.getElementById('trendChart').getContext('2d');
            document.getElementById('chartSubTitle').innerText = `${fromCurr.code} to ${toCurr.code} (${activeDays}D Trend)`;

            const fromRate = currentRates[fromCurr.code] || 1;
            const toRate = currentRates[toCurr.code] || 1;
            const currentSpot = toRate / fromRate;

            // Generate synthetic trend data points ending at currentSpot
            const labels = [];
            const dataPoints = [];
            const now = new Date();

            for(let i = activeDays; i >= 0; i--) {
                const d = new Date();
                d.setDate(now.getDate() - i);
                labels.push(d.toLocaleDateString(undefined, { month: 'short', day: 'numeric' }));
                
                // Add minor random variance to simulate market fluctuations
                const variance = (Math.random() - 0.48) * 0.03 * currentSpot;
                const pointRate = i === 0 ? currentSpot : currentSpot + variance;
                dataPoints.push(parseFloat(pointRate.toFixed(4)));
            }

            const high = Math.max(...dataPoints);
            const low = Math.min(...dataPoints);
            document.getElementById('periodHigh').innerText = `${high} ${toCurr.code}`;
            document.getElementById('periodLow').innerText = `${low} ${toCurr.code}`;

            const isDark = document.documentElement.classList.contains('dark');
            const gridColor = isDark ? 'rgba(255, 255, 255, 0.05)' : 'rgba(0, 0, 0, 0.05)';
            const textColor = isDark ? '#94a3b8' : '#64748b';

            if(chartInstance) chartInstance.destroy();

            chartInstance = new Chart(ctx, {
                type: 'line',
                data: {
                    labels: labels,
                    datasets: [{
                        label: `${fromCurr.code}/${toCurr.code}`,
                        data: dataPoints,
                        borderColor: '#3b82f6',
                        backgroundColor: (context) => {
                            const chart = context.chart;
                            const {ctx, chartArea} = chart;
                            if (!chartArea) return null;
                            const gradient = ctx.createLinearGradient(0, chartArea.top, 0, chartArea.bottom);
                            gradient.addColorStop(0, 'rgba(59, 130, 246, 0.25)');
                            gradient.addColorStop(1, 'rgba(59, 130, 246, 0.0)');
                            return gradient;
                        },
                        borderWidth: 3,
                        fill: true,
                        tension: 0.35,
                        pointRadius: 3,
                        pointHoverRadius: 6
                    }]
                },
                options: {
                    responsive: true,
                    maintainAspectRatio: false,
                    plugins: {
                        legend: { display: false },
                        tooltip: {
                            mode: 'index',
                            intersect: false,
                            backgroundColor: isDark ? '#1e293b' : '#ffffff',
                            titleColor: isDark ? '#f8fafc' : '#0f172a',
                            bodyColor: isDark ? '#cbd5e1' : '#334155',
                            borderColor: isDark ? '#334155' : '#e2e8f0',
                            borderWidth: 1
                        }
                    },
                    scales: {
                        x: {
                            grid: { color: gridColor },
                            ticks: { color: textColor, font: { size: 10 } }
                        },
                        y: {
                            grid: { color: gridColor },
                            ticks: { color: textColor, font: { size: 10 } }
                        }
                    }
                }
            });
        }

        // Event Listeners setup
        amountInput.addEventListener('input', () => {
            calculateConversion();
            updateQuickGrid();
        });

        fromDropdownBtn.addEventListener('click', (e) => {
            e.stopPropagation();

        toDropdownBtn.addEventListener('click', (e) => {

            fromMenu.classList.add('hidden');
            toMenu.classList.toggle('hidden');
            toSearch.focus();
        });

        fromSearch.addEventListener('input', () => renderDropdownLists());
        toSearch.addEventListener('input', () => renderDropdownLists());

        document.addEventListener('click', () => closeDropdowns());
        document.addEventListener('keydown', (e) => {
            if(e.key === 'Escape') closeDropdowns();
        });

        swapBtn.addEventListener('click', () => {
            const temp = fromCurr;
            fromCurr = toCurr;
            toCurr = temp;
            updateDropdownUI('from');
            updateDropdownUI('to');
            calculateConversion();
            updateQuickGrid();
            updateChart();
        });

        refreshRatesBtn.addEventListener('click', () => fetchRates());

        // Timeframe selector buttons
        document.querySelectorAll('.timeframe-btn').forEach(btn => {
            btn.addEventListener('click', (e) => {
                document.querySelectorAll('.timeframe-btn').forEach(b => {
                    b.classList.remove('active', 'bg-white', 'dark:bg-slate-700', 'shadow-sm', 'text-slate-800');
                    b.classList.add('text-slate-500', 'dark:text-slate-400');
                });
                e.target.classList.add('active', 'bg-white', 'dark:bg-slate-700', 'shadow-sm', 'text-slate-800');
                e.target.classList.remove('text-slate-500', 'dark:text-slate-400');
                activeDays = parseInt(e.target.getAttribute('data-days'));
                updateChart();
            });
        });

        // Initialize App
        updateDropdownUI('from');
        updateDropdownUI('to');
        renderDropdownLists();
        fetchRates();
    </script>
